#### 修改4：改进 handleCreateApp 函数（第140-174行）
```typescript
async function handleCreateApp() {
  if (!newApp.value.name.trim()) {
    ElMessage.warning('请输入应用名称')
    return
  }

  creating.value = true
  try {
    if (editingApp.value) {
      // 编辑应用
      await appAPI.update(editingApp.value.id, {
        name: newApp.value.name,
        description: newApp.value.description
      })
      ElMessage.success('应用更新成功')
    } else {
      // 创建应用
      const res = await appAPI.create({
        name: newApp.value.name,
        description: newApp.value.description
      })
      ElMessage.success('应用创建成功')
      // 跳转到应用设计器
      router.push(`/app/app-designer/${res.id}`)
    }
    showCreateDialog.value = false
    newApp.value = { name: '', description: '' }
    editingApp.value = null
    loadApps()
  } catch (error) {
    ElMessage.error(editingApp.value ? '更新失败' : '创建失败')
  } finally {
    creating.value = false
  }
}
```
- 根据 `editingApp` 判断是创建还是编辑
- 编辑时调用 `appAPI.update`
- 创建时调用 `appAPI.create` 并跳转到设计器

#### 修改6：改进 handleCommand 函数（第181-205行）
```typescript
async function handleCommand(command: string, app: any) {
  switch (command) {
    case 'view':
      // 查看应用 - 跳转到应用首页
      router.push(`/app/${app.id}`)
      break
    case 'design':
      router.push(`/app/app-designer/${app.id}`)
      break
    case 'edit':
      // 编辑应用信息
      editApp(app)
      break
    case 'delete':
      try {
        await ElMessageBox.confirm(`确定要删除应用"${app.name}"吗？`, '删除确认', { type: 'warning' })
        await appAPI.delete(app.id)
        ElMessage.success('删除成功')
        loadApps()
      } catch (e: any) {
        if (e !== 'cancel') ElMessage.error('删除失败')
      }
      break
  }
}
```
- 添加 `view` 命令处理，跳转到查看应用页面
- 添加 `edit` 命令处理，打开编辑对话框
- 修复删除功能，真正调用 `appAPI.delete(app.id)`
